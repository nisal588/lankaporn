    <script>
        const videoInput = document.getElementById('videoInput');
        const progressContainer = document.getElementById('progressContainer');
        const progressFill = document.getElementById('progressFill');
        const uploadStatus = document.getElementById('uploadStatus');
        const videoGrid = document.getElementById('videoGrid');

        videoInput.addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            // Show UI progress elements
            progressContainer.style.display = 'block';
            uploadStatus.textContent = `Reading: ${file.name}`;

            // Simulate UI Upload Progress
            let progress = 0;
            const interval = setInterval(() => {
                progress += 10;
                progressFill.style.width = progress + '%';
                
                if (progress >= 100) {
                    clearInterval(interval);
                    uploadStatus.textContent = "Upload complete! (Local simulation)";
                    
                    // Create a local URL to preview the uploaded file instantly in the feed
                    const videoURL = URL.createObjectURL(file);
                    addVideoToFeed(file.name, videoURL);
                    
                    // Reset input after a brief delay
                    setTimeout(() => {
                        progressContainer.style.display = 'none';
                        progressFill.style.width = '0%';
                    }, 2000);
                }
            }, 200);
        });

        function addVideoToFeed(title, src) {
            const card = document.createElement('div');
            card.className = 'video-card';
            card.innerHTML = `
                <video controls src="${src}"></video>
                <div class="video-info">
                    <h3 class="video-title">${title}</h3>
                    <div class="video-meta">Uploaded just now • 0 views</div>
                </div>
            `;
            // Insert new video at the top of the grid
            videoGrid.insertBefore(card, videoGrid.firstChild);
        }
    </script>
